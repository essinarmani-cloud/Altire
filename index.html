        function animateCounters() {

            const counters = document.querySelectorAll('.stat-number');

            counters.forEach(counter => {

                const target = parseInt(counter.dataset.count);

                const suffix = counter.textContent.includes('%') ? '%' :

                    counter.textContent.includes('+') ? '+' : '';

                const duration = 2000;

                const startTime = performance.now();

                function update(currentTime) {

                    const elapsed = currentTime - startTime;

                    const progress = Math.min(elapsed / duration, 1);

                    const easedProgress = 1 - Math.pow(1 - progress, 3);

                    const currentValue = Math.floor(easedProgress * target);

                    counter.textContent = currentValue.toLocaleString('fa-IR') + suffix;

                    if (progress < 1) {

                        requestAnimationFrame(update);

                    } else {

                        counter.textContent = target.toLocaleString('fa-IR') + suffix;

                    }

                }

                requestAnimationFrame(update);

            });

这个页面通过视觉设计营造出专业、可靠的高端轮胎品牌氛围，并提供了直观的购物流程。我们来看看几个核心的设计点：

· 品牌氛围与信任感：整体采用深色背景搭配金色点缀，传递出稳重、高级的质感。3D轮胎模型的金属感和光影效果，强化了产品的专业形象。页面底部的动态数据统计（如“25+年经验”）能快速建立用户信任。

· 沉浸式3D交互体验：首屏的3D轮胎模型是视觉焦点。它不仅是装饰，用户还能拖拽旋转轮胎，从各个角度查看“产品细节”。这种互动方式比静态图片更能抓住注意力，并模拟了线下查看实物的体验。

· 清晰的电商转化路径：页面结构遵循了“吸引注意 → 展示产品 → 促成行动”的逻辑。产品卡片在悬停时有上浮和轮胎旋转的效果，“加入购物车”按钮和侧边栏的购物车系统都设计得醒目且操作流畅，旨在减少用户下单的阻力。
